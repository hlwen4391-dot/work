## 1. 战斗回放时恢复初始HP状态

### 📍 位置：BattleReplayer._restoreInitialState()

**文件**: `system/BattleReplayer.js`

**功能**: 回放时将所有单位的HP恢复到战斗开始时的初始状态

```javascript
/**
 * 恢复初始状态
 * @private
 */
_restoreInitialState(initialState) {
    // 清除所有单位的Buff和状态
    const BuffSystem = require("BuffSystem");
    const BuffComponent = require("BuffComponent");
    const TeamRef = require("TeamRef");

    // 重要：恢复 TeamRef，确保所有单位（包括已死亡的）都在队伍列表中
    if (TeamRef.herosRef) {
        TeamRef.herosRef.length = 0; // 清空数组
    }
    if (TeamRef.monstersRef) {
        TeamRef.monstersRef.length = 0; // 清空数组
    }

    // 重新添加所有单位到 TeamRef（包括已死亡的，因为回放时需要它们）
    Object.values(this.unitMap).forEach(unit => {
        if (unit && unit.isValid) {
            const team = unit.getComponent("TeamComponent");
            if (team) {
                if (team.team === "hero" && TeamRef.herosRef) {
                    if (TeamRef.herosRef.indexOf(unit) === -1) {
                        TeamRef.herosRef.push(unit);
                    }
                } else if (team.team === "monster" && TeamRef.monstersRef) {
                    if (TeamRef.monstersRef.indexOf(unit) === -1) {
                        TeamRef.monstersRef.push(unit);
                    }
                }
            }
        }
    });

    // 清除所有单位的Buff
    Object.values(this.unitMap).forEach(unit => {
        if (unit && unit.isValid) {
            const buffs = unit.getComponents(BuffComponent);
            for (let buff of buffs) {
                BuffSystem.removeBuff(unit, buff.buffName, this.logger.log.bind(this.logger));
            }
        }
    });

    // ⭐ 恢复英雄状态（包括HP）
    if (initialState.heros) {
        initialState.heros.forEach(unitData => {
            const unit = this.unitMap[unitData.id];
            if (unit && unit.isValid) {
                const stats = unit.getComponent("StatsComponent");
                if (stats) {
                    // ⭐ 恢复HP到初始值
                    stats.hp = unitData.hp;
                    stats.maxHp = unitData.maxHp;
                    stats.attack = unitData.attack;
                    stats.defense = unitData.defense;
                    stats.speed = unitData.speed;
                    // 更新血条显示
                    stats.updateHealthBar();
                }
            }
        });
    }

    // ⭐ 恢复怪物状态（包括HP）
    if (initialState.monsters) {
        initialState.monsters.forEach(unitData => {
            const unit = this.unitMap[unitData.id];
            if (unit && unit.isValid) {
                const stats = unit.getComponent("StatsComponent");
                if (stats) {
                    // ⭐ 恢复HP到初始值
                    stats.hp = unitData.hp;
                    stats.maxHp = unitData.maxHp;
                    stats.attack = unitData.attack;
                    stats.defense = unitData.defense;
                    stats.speed = unitData.speed;
                    // 更新血条显示
                    stats.updateHealthBar();
                }
            }
        });
    }

    cc.log("[BattleReplayer] 初始状态已恢复，所有Buff已清除");
}
```

**关键代码**:
```javascript
// 恢复HP到初始值
stats.hp = unitData.hp;        // ⭐ 恢复当前HP
stats.maxHp = unitData.maxHp;  // ⭐ 恢复最大HP
stats.updateHealthBar();       // 更新血条显示
```

**调用时机**:
- 回放开始时调用
- 重新播放时调用（点击播放按钮）

## 2. UI显示支持（治疗效果显示）

### 📍 位置：HealthBar.showDamage()

**文件**: `ecs/HealthBar.js`

**功能**: 支持显示治疗效果（绿色数字，向上飘）

```javascript
/**
 * 显示伤害数字（带向上飘动画）
 * @param {number} value - 伤害值（负数表示伤害，正数表示治疗）
 * @param {string} type - 伤害类型: 'normal'(普通), 'crit'(暴击), 'miss'(闪避), 'heal'(治疗)
 */
showDamage(value, type = 'normal') {
    if (!this.damageLabel) return;

    const startY = this.damageLabel.node.y;
    const startScale = this.damageLabel.node.scale || 1.0;

    switch (type) {
        case 'miss':  // 闪避
            this.damageLabel.string = "MISS";
            this.damageLabel.node.color = cc.color(255, 255, 0);  // 黄色
            this.damageLabel.node.scale = 1.0;
            this._playFloatAnimation(startY, startScale, 20, 1.0, false);
            break;

        case 'heal':  // ⭐ 治疗
            this.damageLabel.string = "+" + Math.floor(value);
            this.damageLabel.node.color = cc.color(50, 205, 50);  // 绿色
            this.damageLabel.node.scale = 1.2;
            this._playFloatAnimation(startY, startScale, 40, 1.0, false);
            break;

        case 'normal':  // 普通伤害
        default:
            this.damageLabel.string = "-" + Math.floor(value);
            this.damageLabel.node.color = cc.color(255, 255, 255);  // 白色
            this.damageLabel.node.scale = 1.0;
            this._playFloatAnimation(startY, startScale, 30, 1.0, false);
            break;
    }
}
```

**关键代码**:
```javascript
case 'heal':  // 治疗
    this.damageLabel.string = "+" + Math.floor(value);
    this.damageLabel.node.color = cc.color(50, 205, 50);  // 绿色
    this.damageLabel.node.scale = 1.2;
    this._playFloatAnimation(startY, startScale, 40, 1.0, false);
    break;
```

**说明**: 
- 这个只是UI显示支持，当前代码中没有实际的治疗技能
- 如果要添加治疗技能，需要在 `SkillConfig.js` 中添加治疗技能，并在 `CombatSystem` 中添加治疗逻辑

## 3. StatsComponent.updateHealthBar() 支持治疗类型

**文件**: `ecs/StatsComponent.js`

```javascript
/**
 * 更新血条显示
 * @param {number} damage - 本次受到的伤害(可选,用于显示伤害数字)
 * @param {string} type - 伤害类型: 'normal'(普通), 'crit'(暴击), 'miss'(闪避), 'heal'(治疗)
 */
updateHealthBar(damage, type = 'normal') {
    if (this.healthBar) {
        // 获取当前护盾值
        let shieldValue = 0;
        const BuffComponent = require("BuffComponent");
        const allBuffs = this.node.getComponents(BuffComponent);
        
        const shieldBuff = allBuffs.find(b => b.buffName === "护盾");
        if (shieldBuff) {
            shieldValue = shieldBuff.shieldValue || 0;
        }

        this.healthBar.updateHealth(this.hp, this.maxHp, shieldValue);
        
        // 显示伤害数字（包括闪避MISS和治疗）
        if (damage !== undefined) {
            if (type === 'miss') {
                this.healthBar.showDamage(0, 'miss');
            } else if (damage > 0) {
                // ⭐ 支持显示治疗（type='heal'时）
                this.healthBar.showDamage(damage, type);
            }
        }
    }
}
```

**说明**: 
- `updateHealthBar()` 方法支持 `type='heal'` 参数
- 但当前代码中没有实际调用 `updateHealthBar(damage, 'heal')` 的地方

## 💡 如何添加治疗技能

如果要添加治疗技能，需要：

### 1. 在 SkillConfig.js 中添加治疗技能

```javascript
// 治疗术 - 恢复目标生命值
healSkill: {
    name: "治疗术",
    id: SkillEnum.healSkill,
    cooldown: 5.0,
    effect: (self, target, log) => {
        log(`💚 ${self.name} 对 ${target.name} 使用治疗术！`);
        return [
            { type: "heal", value: 30 }  // 恢复30点HP
        ];
    }
}
```

### 2. 在 SkillSystem.js 中添加治疗处理

```javascript
useSkill(entity, target, skill, log, rand, recorder) {
    // ... 现有代码 ...

    const events = skill.effect(entity, target, log, rand);

    for (let evt of events) {
        switch (evt.type) {
            case "damage":
                CombatSystem.damage(entity, target, evt.value, log, recorder, rand);
                break;

            case "damageTrue":
                CombatSystem.damageTrue(entity, target, evt.value, log, recorder, rand);
                break;

            case "heal":  // ⭐ 添加治疗处理
                CombatSystem.heal(entity, target, evt.value, log, recorder);
                break;

            // ... 其他事件类型 ...
        }
    }
}
```

### 3. 在 CombatSystem.js 中添加 heal() 方法

```javascript
/**
 * 治疗（恢复生命值）
 * @param {cc.Node} caster - 施法者
 * @param {cc.Node} target - 目标
 * @param {number} healAmount - 治疗量
 * @param {Function} log - 日志函数
 * @param {Object} recorder - 战斗记录器（可选）
 */
heal(caster, target, healAmount, log, recorder) {
    const tgtStats = target.getComponent("StatsComponent");
    if (!tgtStats) return;

    // 计算实际恢复量（不能超过最大HP）
    const actualHeal = Math.min(healAmount, tgtStats.maxHp - tgtStats.hp);
    
    if (actualHeal > 0) {
        // ⭐ 恢复HP
        tgtStats.hp += actualHeal;
        tgtStats.hp = Math.min(tgtStats.hp, tgtStats.maxHp);  // 确保不超过最大HP
        
        // 更新血条显示（使用'heal'类型）
        tgtStats.updateHealthBar(actualHeal, 'heal');
        
        log(`💚 ${caster.name} 对 ${target.name} 恢复了 ${actualHeal} 点生命值`);
        
        // 记录治疗事件（如果recorder支持）
        if (recorder) {
            recorder.recordEvent("heal", {
                casterName: caster.name,
                targetName: target.name,
                healAmount: actualHeal,
                targetHp: tgtStats.hp
            });
        }
    } else {
        log(`💚 ${target.name} 生命值已满，无法治疗`);
    }
}
```
